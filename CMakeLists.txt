cmake_minimum_required(VERSION 3.16)
project(final_project_aim LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
option(FPA_BUILD_HARDWARE "Build Hikrobot/OpenVINO/serial task executables" ON)
option(FPA_ENABLE_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
include(CTest)
find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d imgcodecs videoio highgui dnn)
find_package(Eigen3 REQUIRED)
find_package(fmt REQUIRED)
find_package(spdlog REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)
if(FPA_ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(aim_tools STATIC
  tools/exiter.cpp tools/extended_kalman_filter.cpp tools/img_tools.cpp
  tools/math_tools.cpp tools/plotter.cpp tools/trajectory.cpp tools/logger.cpp tools/crc.cpp)
target_include_directories(aim_tools PUBLIC "${PROJECT_SOURCE_DIR}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(aim_tools PUBLIC Eigen3::Eigen ${OpenCV_LIBS} fmt::fmt
  spdlog::spdlog yaml-cpp nlohmann_json::nlohmann_json Threads::Threads)
add_library(aim_core STATIC
  tasks/auto_aim/armor.cpp tasks/auto_aim/solver.cpp tasks/auto_aim/target.cpp
  tasks/auto_aim/controller.cpp)
target_link_libraries(aim_core PUBLIC aim_tools)
target_compile_options(aim_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(aim_simulator sim/aim_simulator.cpp)
target_link_libraries(aim_simulator PRIVATE aim_core)
if(BUILD_TESTING)
  add_executable(solver_tests tests/test_solver.cpp)
  target_link_libraries(solver_tests PRIVATE aim_core)
  add_test(NAME solver_tests COMMAND solver_tests "${PROJECT_SOURCE_DIR}/sim/camera.yaml")
  add_test(NAME simulator_physics_tests COMMAND aim_simulator --self-test --config "${PROJECT_SOURCE_DIR}/sim/camera.yaml")
  set_tests_properties(simulator_physics_tests PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME simulator_blackbox_tests COMMAND "${Python3_EXECUTABLE}"
      "${PROJECT_SOURCE_DIR}/tests/test_simulator.py" --binary $<TARGET_FILE:aim_simulator>)
    set_tests_properties(simulator_blackbox_tests PROPERTIES TIMEOUT 120)
    add_test(NAME plotter_loopback_tests COMMAND "${Python3_EXECUTABLE}"
      "${PROJECT_SOURCE_DIR}/tests/test_plotter.py" --binary $<TARGET_FILE:aim_simulator>)
    set_tests_properties(plotter_loopback_tests PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE)
  endif()
  if(EXISTS "${PROJECT_SOURCE_DIR}/tests/test_controller_edges.cpp")
    add_executable(controller_edge_tests tests/test_controller_edges.cpp)
    target_link_libraries(controller_edge_tests PRIVATE aim_core)
    add_test(NAME controller_edge_tests COMMAND controller_edge_tests)
    set_tests_properties(controller_edge_tests PROPERTIES TIMEOUT 60)
  endif()
  add_executable(target_tests tests/test_target.cpp)
  target_link_libraries(target_tests PRIVATE aim_core)
  add_test(NAME target_tests COMMAND target_tests)
  if(EXISTS "${PROJECT_SOURCE_DIR}/tests/test_controller.cpp")
    add_executable(controller_tests tests/test_controller.cpp)
    target_link_libraries(controller_tests PRIVATE aim_core)
    add_test(NAME controller_tests COMMAND controller_tests)
  endif()
endif()
if(FPA_BUILD_HARDWARE)
  if(NOT OpenVINO_DIR AND EXISTS "/opt/intel/openvino_2024.6.0/runtime/cmake")
    set(OpenVINO_DIR "/opt/intel/openvino_2024.6.0/runtime/cmake")
  endif()
  find_package(OpenVINO REQUIRED COMPONENTS Runtime)
  add_subdirectory(io)
  target_link_libraries(io PUBLIC aim_tools)
  add_library(aim_detector STATIC tasks/auto_aim/yolo.cpp tasks/auto_aim/yolos/yolov5.cpp)
  target_link_libraries(aim_detector PUBLIC aim_core openvino::runtime)
  if(BUILD_TESTING)
    add_executable(yolo_parser_tests tests/test_yolo_parser.cpp)
    target_link_libraries(yolo_parser_tests PRIVATE aim_detector)
    add_test(NAME yolo_parser_tests COMMAND yolo_parser_tests configs/offline_cpu.yaml)
    set_tests_properties(yolo_parser_tests PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" TIMEOUT 60)
  endif()
  if(EXISTS "${PROJECT_SOURCE_DIR}/src/task_runner.cpp")
    add_library(task_runner STATIC src/task_runner.cpp)
    target_link_libraries(task_runner PUBLIC aim_detector io)
    foreach(task IN ITEMS 1 2 3)
      add_executable(task_${task} src/task_${task}.cpp)
      target_link_libraries(task_${task} PRIVATE task_runner)
    endforeach()
    if(BUILD_TESTING AND Python3_Interpreter_FOUND)
      add_test(NAME task_cli_tests COMMAND "${Python3_EXECUTABLE}"
        "${PROJECT_SOURCE_DIR}/tests/test_task_cli.py" --build-dir "${CMAKE_BINARY_DIR}")
      set_tests_properties(task_cli_tests PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" TIMEOUT 120)
    endif()
  endif()
  if(BUILD_TESTING AND EXISTS "${PROJECT_SOURCE_DIR}/tests/test_gimbal_transport.cpp")
    add_executable(gimbal_transport_tests tests/test_gimbal_transport.cpp)
    target_link_libraries(gimbal_transport_tests PRIVATE io aim_tools util)
    add_test(NAME gimbal_transport_tests COMMAND gimbal_transport_tests)
    set_tests_properties(gimbal_transport_tests PROPERTIES TIMEOUT 20)
  endif()
endif()
